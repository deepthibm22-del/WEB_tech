1. Count vowels in a string
let str = "javascript";
let count = 0;

for (let i = 0; i < str.length; i++) {
    if ("aeiou".includes(str[i])) {
        count++;
    }
}

console.log("Vowels:", count);


Output: Vowels: 3
2. Count consonants in a string
let str = "javascript";
let count = 0;

for (let i = 0; i < str.length; i++) {
    if (str[i] >= 'a' && str[i] <= 'z' && !"aeiou".includes(str[i])) {
        count++;
    }
}

console.log("Consonants:", count);


Output: Consonants: 7
3. Reverse a string
let str = "hello";
let rev = "";

for (let i = str.length - 1; i >= 0; i--) {
    rev = rev + str[i];
}

console.log("Reversed string:", rev);


Output: olleh
4. Check palindrome string
let str = "madam";
let rev = "";

for (let i = str.length - 1; i >= 0; i--) {
    rev = rev + str[i];
}

if (str === rev) {
    console.log("Palindrome");
} else {
    console.log("Not a palindrome");
}
